"""Interrupt only an obsolete source-absent verify worker after committed exports.

Review before execution. --finalize only confirms exit and finishes a receipt
from an existing immutable pre-signal snapshot; it never sends another signal.
"""

from __future__ import annotations

import argparse
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import select
import signal
import time
import uuid

REQUIRED = ("prepare", "export-baseline", "export-combined", "archive")


def sha(path):
    with Path(path).open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def digest(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def exclusive(path, value):
    with Path(path).open("x") as stream:
        json.dump(value, stream, indent=2, sort_keys=True)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    fd = os.open(Path(path).parent, os.O_RDONLY | os.O_DIRECTORY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def prerequisites(root, checkout, *, running):
    report_path = root / "report.json"
    report = json.loads(report_path.read_text())
    if running and report.get("status") != "running":
        raise ValueError("capture is not running; no interruption permitted")
    stages = {entry["stage"]: entry for entry in report["stages"]}
    if len(stages) != len(report["stages"]) or not all(name in stages for name in REQUIRED):
        raise ValueError("capture requires successful prepare, both exports, and archive")
    if running and "verify" in stages:
        raise ValueError("obsolete verification already completed; no interruption permitted")
    if stages["prepare"]["original_native_id"] != stages["prepare"]["copied_native_id"]:
        raise ValueError("source copy native ID changed")
    if os.path.lexists(root / "source-copy.ms") or not (root / "og-data/source-copy.ms").is_dir() or stages["archive"].get("recorded_source_absent") is not True:
        raise ValueError("source-absent archive prerequisite failed")
    script = checkout / "examples/state_codec_qualification.py"
    if sha(script) != report["script_sha256"]:
        raise ValueError("capture script hash differs from report")
    exports = {}
    for name in REQUIRED[1:3]:
        stage = stages[name]
        state_id, representation_id = stage["state_id"], stage["representation_id"]
        if state_id != stages["prepare"]["original_native_id"]:
            raise ValueError("export native ID differs from prepared source")
        for value, prefix in ((state_id, "msutils-logical-hash/v1:"), (representation_id, "sha256:")):
            suffix = value.removeprefix(prefix)
            if not value.startswith(prefix) or len(suffix) != 64 or any(c not in "0123456789abcdef" for c in suffix):
                raise ValueError("invalid export identity")
        manifest_path = root / "store/states" / state_id.split(":")[1] / "representations" / representation_id.split(":")[1] / "representation.json"
        manifest = json.loads(manifest_path.read_text())
        operation = uuid.UUID(manifest["capture_attempt"])
        attempt_path = root / "store/attempts" / f"{operation}.json"
        if Path(stage["attempt"]).resolve(strict=True) != attempt_path.resolve(strict=True):
            raise ValueError("reported export attempt differs from canonical store path")
        attempt = json.loads(attempt_path.read_text())
        if not attempt.get("finished") or attempt.get("phase") != "committed" or attempt.get("operation") != "export" or uuid.UUID(attempt["attempt_id"]) != operation:
            raise ValueError("export is not a completed committed export")
        if any(record.get("state_id") != state_id or record.get("representation_id") != representation_id for record in (manifest, attempt)):
            raise ValueError("export IDs disagree")
        if attempt.get("versions") != manifest.get("versions"):
            raise ValueError("export runtime differs from manifest")
        exports[name] = {"manifest_sha256": sha(manifest_path), "attempt_sha256": sha(attempt_path)}
    return report, {
        "report_sha256": sha(report_path),
        "script_sha256": sha(script),
        "stages_sha256": {name: digest(stages[name]) for name in REQUIRED},
        "exports": exports,
        "recorded_source_absent": True,
    }


def process(pid, proc=Path("/proc")):
    base = proc / str(pid)
    if base.stat().st_uid != os.getuid():
        raise ValueError("process belongs to another user")
    stat = (base / "stat").read_text().rsplit(")", 1)[1].split()
    return {
        "pid": pid,
        "ppid": int(stat[1]),
        "startticks": int(stat[19]),
        "state": stat[0],
        "argv": [part.decode() for part in (base / "cmdline").read_bytes().split(b"\0") if part],
    }


def command_matches(info, script, root, stage):
    argv = info["argv"]
    if len(argv) < 2 or Path(argv[1]) != script:
        return False
    options = argv[2:]
    if len(options) % 2 or any(name not in {"--source", "--workdir", "--block-rows", "--stage"} for name in options[::2]):
        return False
    pairs = dict(zip(options[::2], options[1::2], strict=True))
    if len(pairs) * 2 != len(options):
        return False
    return pairs.get("--workdir") == str(root) and pairs.get("--stage") == stage


def select_processes(root, script, proc=Path("/proc")):
    workers = []
    for entry in proc.iterdir():
        if not entry.name.isdecimal():
            continue
        try:
            info = process(int(entry.name), proc)
        except (FileNotFoundError, ProcessLookupError, PermissionError, ValueError):
            continue
        if command_matches(info, script, root, "verify") and info["state"] != "Z":
            workers.append(info)
    if len(workers) != 1:
        raise ValueError("expected exactly one own verify worker; no signal sent")
    worker = workers[0]
    parent = process(worker["ppid"], proc)
    if not command_matches(parent, script, root, None) or parent["state"] == "Z":
        raise ValueError("verify parent is not the exact capture controller")
    return [worker, parent]


def same_identity(current, expected):
    return all(current[key] == expected[key] for key in ("pid", "ppid", "startticks", "argv"))


def exited(info):
    try:
        current = process(info["pid"])
    except (FileNotFoundError, ProcessLookupError):
        return True
    return current["startticks"] != info["startticks"] or current["state"] == "Z"


def artifact_path(output, suffix):
    return output.with_name(output.name + suffix)


def delivery_marker(args, snapshot):
    """Require durable proof that the pidfd SIGINT call returned successfully."""
    snapshot_path = artifact_path(args.output_receipt, ".pre-signal.json")
    marker_path = artifact_path(args.output_receipt, ".signal-delivered.json")
    if not marker_path.is_file():
        raise ValueError("signal delivery marker is absent; manual diagnosis required, cancellation cannot be claimed")
    marker = json.loads(marker_path.read_text())
    if (
        json.loads(snapshot_path.read_text()) != snapshot
        or marker.get("schema_version") != "shinobi-qualification-signal-delivery/v1"
        or marker.get("snapshot_sha256") != sha(snapshot_path)
        or marker.get("target") != snapshot["processes"][0]
        or marker.get("signal") != int(signal.SIGINT)
        or marker.get("signal_name") != "SIGINT"
        or marker.get("delivery_confirmed") is not True
    ):
        raise ValueError("signal delivery marker differs from immutable snapshot or expected SIGINT")
    timestamp = datetime.fromisoformat(marker["delivered_at"])
    if timestamp.tzinfo is None:
        raise ValueError("signal delivery timestamp must include timezone")
    return {"path": str(marker_path), "sha256": sha(marker_path), "marker": marker}


def finalize(args, snapshot):
    delivery = delivery_marker(args, snapshot)
    processes = snapshot["processes"]
    deadline = time.monotonic() + 45
    while not all(exited(info) for info in processes):
        if time.monotonic() >= deadline:
            raise TimeoutError("capture processes still alive; no further signal sent; rerun --finalize after exit")
        time.sleep(0.2)
    old, evidence = prerequisites(args.capture_root, args.capture_checkout, running=False)
    if (
        old.get("status") != "failed"
        or old.get("failed_stage") != "verify"
        or type(old.get("exit_code")) is not int
        or old["exit_code"] not in {-int(signal.SIGINT), 128 + int(signal.SIGINT)}
    ):
        raise ValueError("capture did not terminate with the expected SIGINT verify receipt")
    for key in ("script_sha256", "stages_sha256", "exports", "recorded_source_absent"):
        if evidence[key] != snapshot["capture_evidence"][key]:
            raise ValueError("capture prerequisites changed during handoff")
    receipt = {
        **snapshot,
        "schema_version": "shinobi-qualification-transition/v1",
        "disposition": "cancelled-obsolete-verification",
        "capture_report_sha256": evidence["report_sha256"],
        "capture_script_sha256": evidence["script_sha256"],
        "post_report_sha256": evidence["report_sha256"],
        "completed_at": datetime.now(timezone.utc).isoformat(),
        "signal_delivery": delivery,
        "processes": [{**info, "exit_confirmed": True} for info in processes],
    }
    if delivery_marker(args, snapshot) != delivery:
        raise ValueError("signal delivery evidence changed during finalization")
    exclusive(args.output_receipt, receipt)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--capture-root", type=Path, required=True)
    parser.add_argument("--capture-checkout", type=Path, required=True)
    parser.add_argument("--output-receipt", type=Path, required=True)
    parser.add_argument("--continuation-path", type=Path, required=True)
    parser.add_argument("--finalize", action="store_true")
    args = parser.parse_args()
    args.capture_root = args.capture_root.resolve(strict=True)
    args.capture_checkout = args.capture_checkout.resolve(strict=True)
    args.output_receipt = args.output_receipt.resolve()
    source = Path(json.loads((args.capture_root / "report.json").read_text())["source"]).resolve(strict=True)
    if any(args.output_receipt.is_relative_to(root) for root in (args.capture_root, args.capture_checkout, source)):
        raise ValueError("handoff receipts must be outside capture, checkout, and original source trees")
    snapshot_path = artifact_path(args.output_receipt, ".pre-signal.json")
    if args.output_receipt.exists():
        raise FileExistsError("final handoff receipt already exists")
    if args.finalize:
        snapshot = json.loads(snapshot_path.read_text())
        if (
            snapshot["capture_root"] != str(args.capture_root)
            or snapshot["capture_checkout"] != str(args.capture_checkout)
            or snapshot["continuation_path"] != str(args.continuation_path)
        ):
            raise ValueError("finalize arguments differ from immutable snapshot")
        finalize(args, snapshot)
        return
    if not hasattr(os, "pidfd_open") or not hasattr(signal, "pidfd_send_signal"):
        raise RuntimeError("operator Python must provide Linux pidfd_open and pidfd_send_signal; no PID-only signalling fallback")
    _, evidence = prerequisites(args.capture_root, args.capture_checkout, running=True)
    script = args.capture_checkout / "examples/state_codec_qualification.py"
    processes = select_processes(args.capture_root, script)
    fds = []
    try:
        for info in processes:
            fd = os.pidfd_open(info["pid"])
            fds.append(fd)
            if not same_identity(process(info["pid"]), info) or select.select([fd], [], [], 0)[0]:
                raise ValueError("process identity changed or exited before signal")
        _, current = prerequisites(args.capture_root, args.capture_checkout, running=True)
        if current != evidence:
            raise ValueError("capture changed before signal")
        snapshot = {
            "schema_version": "shinobi-qualification-handoff-snapshot/v1",
            "captured_at": datetime.now(timezone.utc).isoformat(),
            "capture_root": str(args.capture_root),
            "capture_checkout": str(args.capture_checkout),
            "continuation_path": str(args.continuation_path),
            "reason": "supersede obsolete capture-code verification with final-code complete source-absent verification",
            "pre_report_sha256": evidence["report_sha256"],
            "capture_evidence": evidence,
            "processes": [{**info, "script_sha256": evidence["script_sha256"], "exit_confirmed": False} for info in processes],
            "operator_argv": __import__("sys").argv,
            "operator_script_sha256": sha(__file__),
        }
        exclusive(snapshot_path, snapshot)
        for info, fd in zip(processes, fds, strict=True):
            if not same_identity(process(info["pid"]), info) or select.select([fd], [], [], 0)[0]:
                raise ValueError("process identity changed after snapshot; no signal sent")
        signal.pidfd_send_signal(fds[0], signal.SIGINT)
        exclusive(
            artifact_path(args.output_receipt, ".signal-delivered.json"),
            {
                "schema_version": "shinobi-qualification-signal-delivery/v1",
                "snapshot_sha256": sha(snapshot_path),
                "target": snapshot["processes"][0],
                "signal": int(signal.SIGINT),
                "signal_name": "SIGINT",
                "delivered_at": datetime.now(timezone.utc).isoformat(),
                "delivery_confirmed": True,
            },
        )
        finalize(args, snapshot)
    finally:
        for fd in fds:
            os.close(fd)


if __name__ == "__main__":
    main()
